# -----------------------------------------------------------------------------
# 1. Самая простая схема
# -----------------------------------------------------------------------------

class UserSchema(BaseModel):

# -----------------------------------------------------------------------------

class TypesSchema(BaseModel):

# Поле может быть None
# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------
# 40. field_validator
# Проверка одного поля
# -----------------------------------------------------------------------------

class ValidatorSchema(BaseModel):
    age: int

    @field_validator("age")
    @classmethod
    def validate_age(cls, value):
        if value < 18:
            raise ValueError("18+")
        return value


# -----------------------------------------------------------------------------
# 41. model_validator(before)
# До создания модели
# -----------------------------------------------------------------------------

class BeforeSchema(BaseModel):

    @model_validator(mode="before")
    @classmethod
    def before(cls, data):
        return data


# -----------------------------------------------------------------------------
# 42. model_validator(after)
# После создания модели
# -----------------------------------------------------------------------------

class AfterSchema(BaseModel):
    password: str
    password_repeat: str

    @model_validator(mode="after")
    def validate_passwords(self):
        if self.password != self.password_repeat:
            raise ValueError("Passwords mismatch")
        return self


# -----------------------------------------------------------------------------
# 43. computed_field
# Вычисляемое поле
# -----------------------------------------------------------------------------

class ProductSchema(BaseModel):
    price: int
    count: int

    @computed_field
    @property
    def total(self) -> int:
        return self.price * self.count

# -----------------------------------------------------------------------------
# 49. model_config
# Настройки модели
# -----------------------------------------------------------------------------

class ConfigSchema(BaseModel):

    model_config = ConfigDict(
        from_attributes=True,
        extra="ignore",
        frozen=False,
        validate_assignment=True,
        populate_by_name=True,
    )


# -----------------------------------------------------------------------------
# 50. from_attributes=True
# Аналог orm_mode=True из Pydantic v1
# -----------------------------------------------------------------------------

class UserResponse(BaseModel):

    model_config = ConfigDict(
        from_attributes=True
    )

    id: int
    username: str

class AssignmentSchema(BaseModel):

    UserSchema(

# -----------------------------------------------------------------------------
# 55. Annotated + Field
# Современный рекомендуемый синтаксис
# -----------------------------------------------------------------------------

PositiveInt = Annotated[
    int,
    Field(gt=0)
]


class ProductSchema(BaseModel):
    count: PositiveInt

class UserBase(BaseModel):

class UserCreate(UserBase):

class UserResponse(UserBase):

    email: EmailStr | None = None
    age: int | None = None


# -----------------------------------------------------------------------------
